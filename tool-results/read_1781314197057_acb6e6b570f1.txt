    99→export function ProjectDetailView() {
   100→  const {
   101→    currentUser, projects, selectedProjectId, users,
   102→    setActiveView, setSelectedProjectId, deleteProject,
   103→    completeTask, reviseTask, rejectReview, showAlert, showConfirm,
   104→    updateProject, isEditProjectModalOpen, setIsEditProjectModalOpen,
   105→    editProjectData, setEditProjectData
   106→  } = useAppStore()
   107→
   108→  const project = projects.find(p => p.id === selectedProjectId)
   109→  
   110→  const [isEditDriveOpen, setIsEditDriveOpen] = useState(false)
   111→  const [isRejectDialogOpen, setIsRejectDialogOpen] = useState(false)
   112→  const [rejectReason, setRejectReason] = useState('')
   113→  const [driveForm, setDriveForm] = useState<Record<string, string>>({})
   114→  const [folderUserAccess, setFolderUserAccess] = useState<Record<string, { userId: string; userName: string; download: boolean; upload: boolean }[]>>({})
   115→  const [taskInputs, setTaskInputs] = useState<Record<string, string>>({})
   116→  const [taskVerified, setTaskVerified] = useState<Record<string, boolean>>({})
   117→  const [isUploadingDetailDoc, setIsUploadingDetailDoc] = useState(false)
   118→  const [revisionTaskId, setRevisionTaskId] = useState<string | null>(null)
   119→  
   120→  // State for multiple publish links per task
   121→  const [taskPublishLinks, setTaskPublishLinks] = useState<Record<string, PublishLink[]>>({})
   122→
   123→  // Calculate initial task inputs using useMemo
   124→  const initialTaskState = useMemo(() => {
   125→    if (!project) return { inputs: {}, verified: {}, publishLinks: {} }
   126→    const inputs: Record<string, string> = {}
   127→    const verified: Record<string, boolean> = {}
   128→    const publishLinks: Record<string, PublishLink[]> = {}
   129→    
   130→    project.tasks.forEach(t => {
   131→      inputs[t.id] = t.data?.link || ''
   132→      verified[t.id] = false
   133→      
   134→      // Initialize publish links from existing data
   135→      if (t.data?.publishLinks && Array.isArray(t.data.publishLinks)) {
   136→        publishLinks[t.id] = t.data.publishLinks
   137→      } else {
   138→        publishLinks[t.id] = []
   139→      }
   140→    })
   141→    return { inputs, verified, publishLinks }
   142→  }, [project])
   143→
   144→  // Use initial values directly
   145→  const currentTaskInputs = Object.keys(taskInputs).length === 0 ? initialTaskState.inputs : taskInputs
   146→  const currentTaskVerified = Object.keys(taskVerified).length === 0 ? initialTaskState.verified : taskVerified
   147→  const currentTaskPublishLinks = Object.keys(taskPublishLinks).length === 0 ? initialTaskState.publishLinks : taskPublishLinks
   148→
   149→  if (!project) return null
   150→
   151→  const isManagerOrAdmin = currentUser ? ['Manager', 'Admin'].includes(currentUser.role) : false
   152→  const isAdministratorOrAdmin = currentUser ? ['Administrator', 'Admin'].includes(currentUser.role) : false
   153→  const canManageProject = isManagerOrAdmin
   154→
   155→  const formatDateTime = (dateString: string) => {
   156→    if (!dateString) return '-'
   157→    const d = new Date(dateString)
   158→    return d.toLocaleString('id-ID', { dateStyle: 'medium', timeStyle: 'short' })
   159→  }
   160→
   161→  const getUserDetails = (userId: string | null) => {
   162→    const user = users.find(u => u.id === userId)
   163→    return user ? { name: user.name, avatar: user.avatar } : { name: '', avatar: '' }
   164→  }
   165→
   166→  const handleOpenEditDrive = () => {
   167→    const formState: Record<string, string> = {}
   168→    const accessState: Record<string, { userId: string; userName: string; download: boolean; upload: boolean }[]> = {}
   169→    project.driveFolders.forEach(f => {
   170→      formState[f.id] = f.link
   171→      // Initialize from existing assignedUsers data
   172→      if (f.assignedUsers && Array.isArray(f.assignedUsers)) {
   173→        accessState[f.id] = f.assignedUsers.map(u => ({
   174→          userId: u.userId,
   175→          userName: u.userName,
   176→          download: u.download,
   177→          upload: u.upload
   178→        }))
   179→      } else {
   180→        accessState[f.id] = []
   181→      }
   182→    })
   183→    setDriveForm(formState)
   184→    setFolderUserAccess(accessState)
   185→    setIsEditDriveOpen(true)
   186→  }
   187→
   188→  const toggleFolderUserAccess = (folderId: string, userId: string, userName: string, field: 'download' | 'upload') => {
   189→    setFolderUserAccess(prev => {
   190→      const current = prev[folderId] || []
   191→      const existing = current.find(u => u.userId === userId)
   192→      if (existing) {
   193→        return { ...prev, [folderId]: current.map(u => u.userId === userId ? { ...u, [field]: !u[field] } : u) }
   194→      } else {
   195→        return { ...prev, [folderId]: [...current, { userId, userName, download: field === 'download', upload: field === 'upload' }] }
   196→      }
   197→    })
   198→  }
   199→
   200→  const handleSaveDriveLinks = async (e: React.FormEvent) => {
   201→    e.preventDefault()
   202→    // Get unique team members from tasks
   203→    const teamMembers = Array.from(
   204→      new Map(project.tasks.map(t => [t.assignedTo, t])).values()
   205→    ).filter(t => t.assignedTo)
   206→
   207→    const folders = project.driveFolders.map(f => {
   208→      const accessList = folderUserAccess[f.id] || []
   209→      // Build assignedUsers: include all team members that have any access checked
   210→      const assignedUsers = teamMembers
   211→        .filter(t => t.assignedTo)
   212→        .map(t => {
   213→          const existing = accessList.find(u => u.userId === t.assignedTo)
   214→          return {
   215→            userId: t.assignedTo!,
   216→            userName: getUserDetails(t.assignedTo!).name,
   217→            download: existing?.download || false,
   218→            upload: existing?.upload || false
   219→          }
   220→        })
   221→        .filter(u => u.download || u.upload)
   222→      
   223→      return {
   224→        id: f.id,
   225→        link: driveForm[f.id] || f.link,
   226→        assignedUsers
   227→      }
   228→    })
   229→    
   230→    try {
   231→      await fetch('/api/drive-folders', {
   232→        method: 'PUT',
   233→        headers: { 'Content-Type': 'application/json' },
   234→        body: JSON.stringify({ projectId: project.id, folders })
   235→      })
   236→      
   237→      const updatedFolders = project.driveFolders.map(f => {
   238→        const savedFolder = folders.find(sf => sf.id === f.id)
   239→        return {
   240→          ...f,
   241→          link: driveForm[f.id] || f.link,
   242→          assignedUsers: savedFolder?.assignedUsers || []
   243→        }
   244→      })
   245→      updateProject({ ...project, driveFolders: updatedFolders })
   246→      setIsEditDriveOpen(false)
   247→    } catch {
   248→      showAlert('Gagal menyimpan tautan drive')
   249→    }
   250→  }
   251→
   252→  const handleDeleteProject = () => {
   253→    showConfirm(
   254→      'Peringatan: Yakin ingin menghapus proyek ini secara permanen? Aksi ini tidak dapat dibatalkan.',
   255→      async () => {
   256→        try {
   257→          await fetch(`/api/projects?id=${project.id}`, { method: 'DELETE' })
   258→          deleteProject(project.id)
   259→        } catch {
   260→          showAlert('Gagal menghapus proyek')
   261→        }
   262→      }
   263→    )
   264→  }
   265→
   266→  const handleTaskComplete = async (taskId: string, taskData: { link?: string; publishLinks?: PublishLink[] }) => {
   267→    try {
   268→      const response = await fetch('/api/tasks', {
   269→        method: 'PUT',
   270→        headers: { 'Content-Type': 'application/json' },
   271→        body: JSON.stringify({
   272→          projectId: project.id,
   273→          taskId,
   274→          taskData,
   275→          isRevision: revisionTaskId === taskId
   276→        })
   277→      })
   278→      
   279→      if (response.ok) {
   280→        if (revisionTaskId === taskId) {
   281→          reviseTask(project.id, taskId, taskData)
   282→          setRevisionTaskId(null)
   283→        } else {
   284→          completeTask(project.id, taskId, taskData)
   285→        }
   286→      } else {
   287→        showAlert('Gagal menyelesaikan tugas')
   288→      }
   289→    } catch {
   290→      showAlert('Terjadi kesalahan')
   291→    }
   292→  }
   293→
   294→  const handleReviewReject = async () => {
   295→    // Buka dialog input alasan terlebih dahulu
   296→    setIsRejectDialogOpen(true)
   297→    setRejectReason('')
   298→  }
   299→
   300→  const handleConfirmReject = async () => {
   301→    try {
   302→      const response = await fetch('/api/tasks', {
   303→        method: 'PUT',
   304→        headers: { 'Content-Type': 'application/json' },
   305→        body: JSON.stringify({
   306→          projectId: project.id,
   307→          isReviewReject: true,
   308→          rejectReason: rejectReason.trim()
   309→        })
   310→      })
   311→      
   312→      if (response.ok) {
   313→        rejectReview(project.id)
   314→        setIsRejectDialogOpen(false)
   315→        setRejectReason('')
   316→      } else {
   317→        showAlert('Gagal menolak review')
   318→      }
   319→    } catch {
   320→      showAlert('Terjadi kesalahan')
   321→    }
   322→  }
   323→
   324→  const handleSaveEditedProject = async (e: React.FormEvent) => {
   325→    e.preventDefault()
   326→    if (!editProjectData) return
   327→    
   328→    try {
   329→      await fetch('/api/projects', {
   330→        method: 'PUT',
   331→        headers: { 'Content-Type': 'application/json' },
   332→        body: JSON.stringify(editProjectData)
   333→      })
   334→      updateProject(editProjectData)
   335→      setIsEditProjectModalOpen(false)
   336→    } catch {
   337→      showAlert('Gagal menyimpan perubahan')
   338→    }
   339→  }
   340→
   341→  // Upload document from project detail view (Manager only)
   342→  const handleDocUploadFromDetail = async (e: React.ChangeEvent<HTMLInputElement>) => {
   343→    const files = e.target.files
   344→    if (!files || files.length === 0) return
   345→
   346→    setIsUploadingDetailDoc(true)
   347→
   348→    for (const file of Array.from(files)) {
   349→      try {
   350→        const formData = new FormData()
   351→        formData.append('file', file)
   352→        formData.append('projectId', project.id)
   353→        formData.append('label', 'Dokumen Pendukung')
   354→
   355→        const res = await fetch('/api/projects/upload-document', {
   356→          method: 'POST',
   357→          body: formData,
   358→        })
   359→
   360→        if (res.ok) {
   361→          const data = await res.json()
   362→          if (data.success && data.document) {
   363→            const updatedDocs = [...(project.documents || []), data.document]
   364→            updateProject({ ...project, documents: updatedDocs })
   365→          }
   366→        } else {
   367→          showAlert('Gagal mengunggah: ' + file.name)
   368→        }
   369→      } catch (err) {
   370→        console.error('[DOC UPLOAD DETAIL] Failed:', err)
   371→        showAlert('Terjadi kesalahan saat mengunggah dokumen.')
   372→      }
   373→    }
   374→
   375→    setIsUploadingDetailDoc(false)
   376→    e.target.value = '' // Reset input so same file can be selected again
   377→  }
   378→
   379→  // Delete document from project detail view (Manager only)
   380→  const handleDeleteDocument = async (documentId: string) => {
   381→    showConfirm('Yakin ingin menghapus dokumen ini?', async () => {
   382→      try {
   383→        const res = await fetch(`/api/projects/upload-document?projectId=${project.id}&documentId=${documentId}`, {
   384→          method: 'DELETE',
   385→        })
   386→
   387→        if (res.ok) {
   388→          const updatedDocs = (project.documents || []).filter(d => d.id !== documentId)
   389→          updateProject({ ...project, documents: updatedDocs })
   390→        } else {
   391→          showAlert('Gagal menghapus dokumen.')
   392→        }
   393→      } catch {
   394→        showAlert('Terjadi kesalahan saat menghapus dokumen.')
   395→      }
   396→    })
   397→  }
   398→
   399→  const visibleFolders = project.driveFolders.filter(folder => {
   400→    // For managers/admins, show all folders
   401→    if (canManageProject) return true
   402→    const myId = currentUser?.id || ''
   403→    const myRole = currentUser?.role || ''
   404→    // Detect subfolders (parentFolderId or folderId pattern)
   405→    const isSub = folder.parentFolderId || 
   406→      (folder.folderId.includes('-') && ['raw', 'revised', 'final', 'desain', 'lainnya'].some(b => folder.folderId.startsWith(b + '-')))
   407→    if (isSub) {
   408→      // Check assignedUsers first (most precise) — match by userId
   409→      if (folder.assignedUsers?.some((au: any) => au.userId === myId)) return true
   410→      // Then check assignedRoles
   411→      if (!folder.assignedRoles || folder.assignedRoles.length === 0) return true
   412→      return folder.assignedRoles.includes(myRole)
   413→    }
   414→    // For parent folders:
   415→    // 1. Check assignedUsers with download:true or upload:true (manager DL/UL checkboxes)
   416→    if (folder.assignedUsers?.some((au: any) => au.userId === myId && (au.download || au.upload))) return true
   417→    // 2. Check assignedRoles
   418→    if (folder.assignedRoles?.includes(myRole)) return true
   419→    // 3. No restrictions = visible to all
   420→    if (!folder.assignedRoles || folder.assignedRoles.length === 0) {
   421→      // Still respect assignedUsers — if there ARE assignedUsers but none match, hide
   422→      if (folder.assignedUsers && folder.assignedUsers.length > 0) {
   423→        return folder.assignedUsers.some((au: any) => au.userId === myId)
   424→      }
   425→      return true
   426→    }
   427→    return false
   428→  })
   429→
   430→  // Separate parent folders and subfolders (including nested output subfolders)
   431→  const parentFolders = visibleFolders.filter(f => !f.parentFolderId && !(['raw', 'revised', 'final', 'desain', 'lainnya'].some(b => f.folderId.startsWith(b + '-') && f.folderId.includes('-'))))
   432→  const subfolders = visibleFolders.filter(f => f.parentFolderId || (f.folderId.includes('-') && ['raw', 'revised', 'final', 'desain', 'lainnya'].some(b => f.folderId.startsWith(b + '-'))))

   434→  // Get direct child subfolders for a given parent ID (supports 3-level hierarchy)
   435→  const getSubfolders = (parentId: string) => {
   436→    return subfolders.filter(s => {
   437→      // Direct parent match (most reliable)
   438→      if (s.parentFolderId === parentId) return true
   439→      // Legacy: folderId prefix match for top-level folders like "raw"
   440→      if (!s.parentFolderId && s.folderId.startsWith(parentId + '-') && !s.folderId.includes('-output-')) return true
   441→      return false
   442→    })
   443→  }
   444→  
   445→  // Check if a subfolder has nested output subfolders inside it
   446→  const getOutputSubfolders = (userSubfolderId: string) => {
   447→    return subfolders.filter(s => s.parentFolderId === userSubfolderId && s.folderId.includes('-output-'))
   448→  }
   449→
   450→  const visibleTasks = project.tasks.filter(t => 
   451→    isManagerOrAdmin ? true : t.assignedTo === currentUser?.id
   452→  )
   453→
   454→  // Add/Remove publish link handlers
   455→  const addPublishLink = (taskId: string) => {
   456→    const newLink: PublishLink = {
   457→      id: `link-${Date.now()}`,
   458→      platform: 'website',
   459→      url: ''
   460→    }
   461→    setTaskPublishLinks(prev => ({
   462→      ...prev,
   463→      [taskId]: [...(prev[taskId] || []), newLink]
   464→    }))
   465→  }
   466→
   467→  const removePublishLink = (taskId: string, linkId: string) => {
   468→    setTaskPublishLinks(prev => ({
   469→      ...prev,
   470→      [taskId]: (prev[taskId] || []).filter(l => l.id !== linkId)
   471→    }))
   472→  }
   473→
   474→  const updatePublishLink = (taskId: string, linkId: string, field: 'platform' | 'url', value: string) => {
   475→    setTaskPublishLinks(prev => ({
   476→      ...prev,
   477→      [taskId]: (prev[taskId] || []).map(l => 
   478→        l.id === linkId ? { ...l, [field]: value } : l
   479→      )
   480→    }))
   481→  }
   482→
   483→  return (
   484→    <div className="max-w-6xl mx-auto space-y-6">
   485→      {/* Header Actions */}
   486→      <div className="flex flex-col sm:flex-row justify-between items-start sm:items-center gap-4">
   487→        <Button
   488→          variant="ghost"
   489→          onClick={() => setActiveView('dashboard')}
   490→          className="text-stone-500 hover:text-indigo-600 gap-1"
   491→        >
   492→          <ArrowLeft className="w-4 h-4 rotate-180" />
   493→          <span>Kembali ke Dashboard</span>
   494→        </Button>
   495→        
   496→        {canManageProject && (
   497→          <div className="flex gap-2 sm:gap-3">
   498→            <Button
   499→              variant="outline"
   500→              onClick={() => { setEditProjectData(project); setIsEditProjectModalOpen(true); }}
   501→              className="gap-2"
   502→            >
   503→              <Edit className="w-4 h-4" />
   504→              <span className="hidden sm:inline">Edit Proyek</span>
   505→            </Button>
   506→            <Button
   507→              variant="outline"
   508→              onClick={handleDeleteProject}
   509→              className="gap-2 text-red-600 hover:text-red-700 hover:bg-red-50 border-red-200"
   510→            >
   511→              <Trash2 className="w-4 h-4" />
   512→              <span className="hidden sm:inline">Hapus Proyek</span>
   513→            </Button>
   514→          </div>
   515→        )}
   516→      </div>
   517→
   518→      {/* Project Header */}
   519→      <Card>
   520→        <CardContent className="p-4 sm:p-6 lg:p-8">
   521→          <div className="flex flex-col md:flex-row md:justify-between md:items-start gap-4">
   522→            <div className="min-w-0">
   523→              <h2 className="text-2xl sm:text-3xl font-bold text-stone-900 mb-2 break-words">{project.title}</h2>
   524→              <div className="flex flex-wrap items-center gap-2 text-sm text-stone-500">
   525→                <Badge variant="outline" className="font-mono text-xs">{project.id}</Badge>
   526→                <span>•</span>
   527→                <span>Pemohon: <strong className="text-stone-700">{project.requesterUnit}</strong></span>
   528→              </div>
   529→            </div>
   530→            <div className="bg-white px-5 py-3 rounded-2xl border border-stone-200 shadow-sm text-center md:text-right shrink-0">
   531→              <span className="text-[10px] uppercase tracking-wider text-stone-400 font-bold block mb-1">
   532→                Status Saat Ini
   533→              </span>
   534→              <span className="font-bold text-lg text-indigo-700">
   535→                Tahap {project.currentStage}: {STAGES[project.currentStage]}
   536→              </span>
   537→              {project.isFastTrack && (
   538→                <Badge className="bg-amber-500 text-white text-[10px] px-2 py-0.5 ml-2 border-0">
   539→                  <Zap className="h-3 w-3 mr-1" />FAST TRACK
   540→                </Badge>
